# 1. Preparar a release candidata (Turtle - 485 conceitos)
docker compose exec api python -m libris.modules.bibliographic.lc_languages_cli prepare \
  data/vocabularies/lc/languages.skosrdf.ttl \
  --format turtle \
  --release-key lc-languages-skos-2026-10 \

# 2. Ativar a release no catálogo

  --release-key lc-languages-skos-2026-10
